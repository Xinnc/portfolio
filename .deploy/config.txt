name: deploy
services:
  backend:
    build:
      context: /home/xinnc/valeriaPortfolio
      dockerfile: .deploy/php.DockerFile
    container_name: backend
    networks:
      network: null
    restart: unless-stopped
    volumes:
      - type: bind
        source: /home/xinnc/valeriaPortfolio/backend
        target: /var/www/backend
        bind:
          create_host_path: true
      - type: bind
        source: /home/xinnc/valeriaPortfolio/backend/php-custom.ini
        target: /usr/local/etc/php/conf.d/custom.ini
        bind:
          create_host_path: true
      - type: bind
        source: /home/xinnc/valeriaPortfolio/backend/storage
        target: /var/www/backend/storage
        bind:
          create_host_path: true
    working_dir: /var/www/backend
  db:
    container_name: mysql_database
    environment:
      MYSQL_DATABASE: portfolio_landing
      MYSQL_PASSWORD: admin
      MYSQL_ROOT_PASSWORD: admin
      MYSQL_USER: admin
    image: mysql:8.0.23
    networks:
      network: null
    ports:
      - mode: ingress
        target: 3306
        published: "3307"
        protocol: tcp
    restart: unless-stopped
    volumes:
      - type: volume
        source: mysql_data
        target: /var/lib/mysql
        volume: {}
  frontend:
    build:
      context: /home/xinnc/valeriaPortfolio
      dockerfile: .deploy/vue.DockerFile
    container_name: frontend
    networks:
      network: null
    volumes:
      - type: bind
        source: /home/xinnc/valeriaPortfolio/frontend/dist
        target: /app/dist
        bind:
          create_host_path: true
  nginx:
    container_name: nginx
    depends_on:
      backend:
        condition: service_started
        required: true
      db:
        condition: service_started
        required: true
    image: nginx:alpine
    networks:
      network: null
    ports:
      - mode: ingress
        target: 80
        published: "80"
        protocol: tcp
      - mode: ingress
        target: 443
        published: "443"
        protocol: tcp
    volumes:
      - type: bind
        source: /home/xinnc/valeriaPortfolio/.deploy/default.conf
        target: /etc/nginx/conf.d/default.conf
        bind:
          create_host_path: true
      - type: bind
        source: /home/xinnc/valeriaPortfolio/backend
        target: /var/www/backend
        bind:
          create_host_path: true
      - type: bind
        source: /home/xinnc/valeriaPortfolio/frontend/dist
        target: /var/www/frontend/dist
        bind:
          create_host_path: true
networks:
  network:
    name: deploy_network
volumes:
  mysql_data:
    name: deploy_mysql_data
